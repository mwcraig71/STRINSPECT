export default function App() {
  const [view, setView] = useState('inspection'); 
  const [activeNbiItem, setActiveNbiItem] = useState('58');
  const [nomenclature, setNomenclature] = useState(NOMENCLATURES.TXDOT);
  const [inspectionType, setInspectionType] = useState(INSPECTION_TYPES.TOPSIDE);
  const [supportCount] = useState(25);
  const [parsingActive, setParsingActive] = useState(false);
  
  // Sequential Workflow State
  const [pendingFUA, setPendingFUA] = useState(false);

  // Modal Visibility
  const [showCIFModal, setShowCIFModal] = useState(false);
  const [showFUAModal, setShowFUAModal] = useState(false);

  // Sorting & Filtering
  const [sortCriteria, setSortCriteria] = useState('location');
  const [filterType, setFilterType] = useState('All');
  const [rangeMin, setRangeMin] = useState('');
  const [rangeMax, setRangeMax] = useState('');
  const [syncToCurrentLoc, setSyncToCurrentLoc] = useState(false);

  // Form State
  const [editId, setEditId] = useState(null);
  const [currentLocation, setCurrentLocation] = useState('');
  const [element, setElement] = useState(null);
  const [environment, setEnvironment] = useState('2');
  const [defect, setDefect] = useState(null);
  const [conditionState, setConditionState] = useState('CS1');
  const [quantity, setQuantity] = useState('');
  const [maintenanceQuantity, setMaintenanceQuantity] = useState('');
  const [size, setSize] = useState('');
  const [locationDesc, setLocationDesc] = useState('');
  const [isCritical, setIsCritical] = useState(false);
  const [isMaintenance, setIsMaintenance] = useState(false);
  const [photos, setPhotos] = useState([]);
  const [savedDefects, setSavedDefects] = useState([]);
  
  const [nbiRatings, setNbiRatings] = useState(INITIAL_NBI_RATINGS);
  const [showDefectPopover, setShowDefectPopover] = useState(null); 

  // Modal-specific state
  const [cifData, setCifData] = useState({
    structureNumber: '18-061-0081-13-133',
    inspectionDate: '02/15/2026',
    findings: '',
    defectDistress: '',
    material: '',
    referenceFeature: '',
    recommendation: '',
    phoneNotified: false,
    assetWiseLogged: false,
    photos: []
  });

  const [fuaData, setFuaData] = useState({
    fuaId: '645238',
    priority: 'Level 3', 
    description: '',
    recommendation: '',
    phoneNotified: false,
    assetWiseLogged: false,
    photos: []
  });

  // --- Derived State & Logic ---

  const activeNbiData = useMemo(() => nbiRatings.find(r => r.item === activeNbiItem), [nbiRatings, activeNbiItem]);
  const activeNbiIdx = useMemo(() => nbiRatings.findIndex(r => r.item === activeNbiItem), [nbiRatings, activeNbiItem]);
  
  const locationSequence = useMemo(() => {
    const sequence = [];
    if (inspectionType === INSPECTION_TYPES.TOPSIDE) {
      sequence.push('Approach 1');
      for (let i = 1; i < supportCount; i++) {
        sequence.push(`Joint ${i}`, `Span ${i}`);
      }
      sequence.push('Approach 2');
    } else {
      const start = nomenclature === NOMENCLATURES.TXDOT ? 'Abutment 1' : 'End Bent 1';
      const end = nomenclature === NOMENCLATURES.TXDOT ? `Abutment ${supportCount}` : 'End Bent 2';
      sequence.push(start);
      for (let i = 1; i < supportCount - 1; i++) {
        sequence.push(`Span ${i}`, `Bent ${nomenclature === NOMENCLATURES.TXDOT ? i + 1 : i}`);
      }
      sequence.push(`Span ${supportCount - 1}`, end);
    }
    return sequence;
  }, [nomenclature, inspectionType, supportCount]);

  const filteredElements = useMemo(() => {
    if (!currentLocation) return [];
    if (currentLocation.includes('Joint')) return SNBI_ELEMENTS.filter(e => e.category === 'Joint');
    if (currentLocation.includes('Approach')) return SNBI_ELEMENTS.filter(e => ['Deck', 'Railing'].includes(e.category));
    if (currentLocation.includes('Span')) return SNBI_ELEMENTS.filter(e => ['Deck', 'Superstructure', 'Railing'].includes(e.category));
    if (currentLocation.includes('Abutment') || currentLocation.includes('Bent') || currentLocation.includes('End Bent')) {
      return SNBI_ELEMENTS.filter(e => ['Substructure', 'Bearing'].includes(e.category));
    }
    return SNBI_ELEMENTS;
  }, [currentLocation]);

  const isDefectValid = (elementId, defectId) => (DEFECTS_BY_ELEMENT[elementId] || []).some(d => d.id === defectId);

  const applyFilters = (list) => {
    let result = [...list];
    if (syncToCurrentLoc) {
      result = result.filter(d => d.location === currentLocation);
    } else {
      if (filterType !== 'All') result = result.filter(d => d.location.includes(filterType));
      if (rangeMin || rangeMax) {
        result = result.filter(d => {
          const numMatch = String(d.location).match(/\d+/);
          if (!numMatch) return true;
          const num = parseInt(numMatch[0]);
          const min = rangeMin === '' ? -Infinity : parseInt(rangeMin);
          const max = rangeMax === '' ? Infinity : parseInt(rangeMax);
          return num >= min && num <= max;
        });
      }
    }
    return result.sort((a, b) => {
      if (sortCriteria === 'location') return String(a.location).localeCompare(String(b.location), undefined, { numeric: true });
      if (sortCriteria === 'severity') {
        const order = { 'CS4': 0, 'CS3': 1, 'CS2': 2, 'CS1': 3 };
        return (order[a.cs] ?? 4) - (order[b.cs] ?? 4);
      }
      return String(a.element).localeCompare(String(b.element));
    });
  };

  const legacyManifest = useMemo(() => {
    return applyFilters(savedDefects.filter(d => d.needsVerification || d.isLegacy));
  }, [savedDefects, filterType, rangeMin, rangeMax, syncToCurrentLoc, currentLocation, sortCriteria]);

  const sessionManifest = useMemo(() => {
    return applyFilters(savedDefects.filter(d => !d.needsVerification && !d.isLegacy));
  }, [savedDefects, filterType, rangeMin, rangeMax, syncToCurrentLoc, currentLocation, sortCriteria]);

  const elementSummary = useMemo(() => {
    const summary = {};
    savedDefects.forEach(d => {
      const elKey = d.elementId || d.element;
      if (!summary[elKey]) {
        const fullEl = SNBI_ELEMENTS.find(e => e.id === d.elementId || e.name === d.element);
        summary[elKey] = { 
          name: String(fullEl?.name || d.element), 
          unit: String(fullEl?.unit || ''), 
          CS1: 0, CS2: 0, CS3: 0, CS4: 0, 
          maintQty: 0, 
          critQty: 0,
          total: 0 
        };
      }
      const qty = parseFloat(d.quantityValue) || 0;
      const mQty = parseFloat(d.maintenanceQuantityValue) || qty;
      
      summary[elKey][d.cs] += qty;
      summary[elKey].total += qty;
      if (d.isMaintenance) summary[elKey].maintQty += mQty;
      if (d.isCritical) summary[elKey].critQty += qty;
    });
    return Object.values(summary);
  }, [savedDefects]);

  const criticalFindingsSummary = useMemo(() => savedDefects.filter(d => d.isCritical), [savedDefects]);
  const maintenanceSummary = useMemo(() => savedDefects.filter(d => d.isMaintenance), [savedDefects]);

  // --- Handlers ---

  useEffect(() => {
    if (!editId) setCurrentLocation(locationSequence[0]);
  }, [locationSequence, editId]);

  useEffect(() => {
    if (filteredElements.length > 0 && !editId) setElement(filteredElements[0]);
  }, [filteredElements, editId]);

  useEffect(() => {
    if (element && !editId) {
      const available = DEFECTS_BY_ELEMENT[element.id] || [];
      setDefect(available[0] || null);
    }
  }, [element, editId]);

  const handlePhotoUpload = (e, target = 'main') => {
    const files = Array.from(e.target.files);
    const newPhotos = files.map(file => ({ 
      url: URL.createObjectURL(file), 
      name: file.name,
      description: '' 
    }));
    if (target === 'cif') setCifData(prev => ({ ...prev, photos: [...prev.photos, ...newPhotos] }));
    else if (target === 'fua') setFuaData(prev => ({ ...prev, photos: [...prev.photos, ...newPhotos] }));
    else setPhotos([...photos, ...newPhotos]);
  };

  const updatePhotoDescription = (idx, desc, target = 'main') => {
    if (target === 'cif') {
      const updated = [...cifData.photos];
      updated[idx].description = desc;
      setCifData({ ...cifData, photos: updated });
    } else if (target === 'fua') {
      const updated = [...fuaData.photos];
      updated[idx].description = desc;
      setFuaData({ ...fuaData, photos: updated });
    } else {
      const updated = [...photos];
      updated[idx].description = desc;
      setPhotos(updated);
    }
  };

  const copyNoteToPhoto = (idx, target = 'main') => {
    const notePrefix = "previous note: ";
    if (target === 'cif') {
      const updated = [...cifData.photos];
      updated[idx].description = notePrefix + cifData.findings;
      setCifData({ ...cifData, photos: updated });
    } else if (target === 'fua') {
      const updated = [...fuaData.photos];
      updated[idx].description = notePrefix + fuaData.description;
      setFuaData({ ...fuaData, photos: updated });
    } else {
      const updated = [...photos];
      updated[idx].description = notePrefix + locationDesc;
      setPhotos(updated);
    }
  };

  const removePhoto = (idx, target = 'main') => {
    if (target === 'cif') setCifData(prev => ({ ...prev, photos: prev.photos.filter((_, i) => i !== idx) }));
    else if (target === 'fua') setFuaData(prev => ({ ...prev, photos: prev.photos.filter((_, i) => i !== idx) }));
    else setPhotos(photos.filter((_, i) => i !== idx));
  };

  const handleSave = () => {
    const record = {
      id: editId || Date.now(),
      location: String(currentLocation),
      elementId: String(element?.id || ''),
      element: String(element?.name || ''),
      environment: String(environment),
      defect: String(defect?.name || ''),
      defectId: String(defect?.id || ''),
      cs: String(conditionState),
      quantityValue: String(quantity),
      maintenanceQuantityValue: String(maintenanceQuantity || quantity),
      quantity: `${quantity} ${defect?.unit || ''}`,
      size: String(size),
      locationDesc: String(locationDesc),
      photosCount: photos.length,
      photos: [...photos],
      isCritical,
      isMaintenance,
      needsVerification: false,
      isLegacy: false
    };

    if (editId) {
      setSavedDefects(savedDefects.map(d => d.id === editId ? record : d));
      setEditId(null);
    } else {
      setSavedDefects([record, ...savedDefects]);
    }
    
    if (isCritical) {
      setCifData(prev => ({
        ...prev,
        findings: locationDesc,
        defectDistress: defect?.name || '',
        material: element?.material || '',
        referenceFeature: `${element?.id || ''}. ${element?.name || ''}`,
        photos: [...photos]
      }));
      setPendingFUA(isMaintenance);
      setShowCIFModal(true);
    } else if (isMaintenance) {
      setFuaData(prev => ({
        ...prev,
        description: locationDesc,
        bridgeComponent: element?.name || '',
        photos: [...photos]
      }));
      setShowFUAModal(true);
    }

    setQuantity(''); setMaintenanceQuantity(''); setSize(''); setLocationDesc(''); setIsCritical(false); setIsMaintenance(false); setPhotos([]);
  };

  const startEdit = (record) => {
    setEditId(record.id);
    setView('inspection');
    setCurrentLocation(record.location);
    const matchedEl = SNBI_ELEMENTS.find(e => String(e.id) === String(record.elementId));
    setElement(matchedEl);
    setConditionState(record.cs);
    setQuantity(record.quantityValue);
    setMaintenanceQuantity(record.maintenanceQuantityValue || '');
    setSize(record.size);
    setLocationDesc(record.locationDesc);
    setIsCritical(record.isCritical || false);
    setIsMaintenance(record.isMaintenance || false);
    setPhotos(record.photos || []);
    const available = DEFECTS_BY_ELEMENT[matchedEl?.id] || [];
    setDefect(available.find(d => String(d.id) === String(record.defectId)) || available[0]);
    window.scrollTo({ top: 0, behavior: 'smooth' });
  };

  const deleteDefect = (id) => {
    if (window.confirm("Delete diagnostic record?")) {
      setSavedDefects(savedDefects.filter(d => d.id !== id));
    }
  };

  const verifyDefect = (id) => {
    setSavedDefects(savedDefects.map(d => d.id === id ? { ...d, needsVerification: false, isLegacy: true } : d));
  };

  const updateSubComponent = (itemIndex, compIndex, field, value) => {
    setNbiRatings(prev => {
      const next = [...prev];
      const nextItem = { ...next[itemIndex] };
      const nextSub = [...nextItem.subComponents];
      nextSub[compIndex] = { ...nextSub[compIndex], [field]: value };
      nextItem.subComponents = nextSub;
      next[itemIndex] = nextItem;
      return next;
    });
  };

  const simulateReportParsing = () => {
    setParsingActive(true);
    const ts = Date.now();
    setTimeout(() => {
      const legacyElements = [
        { id: `leg-${ts}`, location: 'Span 1', elementId: '107', element: 'Steel Girder', environment: '2', defect: 'Corrosion/Section Loss', defectId: 'corr', cs: 'CS2', quantityValue: '10', quantity: '10 in', size: 'N/A', locationDesc: 'Surface rust.', needsVerification: true, isLegacy: true, photos: [], photosCount: 0 }
      ];
      setSavedDefects([...legacyElements, ...savedDefects]);
      setParsingActive(false);
      setView('inspection');
    }, 1200);
  };

  const completeCIF = () => {
    if (!cifData.phoneNotified || !cifData.assetWiseLogged) {
      alert("Verification of immediate safety protocol execution is mandatory.");
      return;
    }
    setShowCIFModal(false);
    if (pendingFUA) {
      setFuaData(prev => ({
        ...prev,
        description: cifData.findings,
        bridgeComponent: cifData.referenceFeature,
        repairCategory: `Remedial Action: ${cifData.defectDistress}`,
        recommendation: cifData.recommendation,
        photos: [...cifData.photos] 
      }));
      setShowFUAModal(true);
      setPendingFUA(false);
    }
  };

  const completeFUA = () => {
    if ((fuaData.priority === 'Level 1' || fuaData.priority === 'Level 2') && !fuaData.phoneNotified) {
      alert("Priority Levels 1 & 2 require immediate phone notification.");
      return;
    }
    if (!fuaData.assetWiseLogged) {
      alert("AssetWise synchronization must be confirmed.");
      return;
    }
    setShowFUAModal(false);
  };

  const toggleInspectionType = () => {
    setInspectionType(prev => prev === INSPECTION_TYPES.TOPSIDE ? INSPECTION_TYPES.UNDERSIDE : INSPECTION_TYPES.TOPSIDE);
  };

  return (
    <div className="min-h-screen bg-slate-100 flex flex-col font-sans text-slate-900 pb-24">
      <header className="bg-slate-900 text-white p-4 shadow-2xl sticky top-0 z-50">
        <div className="flex items-center justify-between mb-4">
          <h1 className="text-xl font-black tracking-tight flex items-center gap-2 uppercase">
            <Activity className="text-blue-400" size={26} /> 
            {view === 'inspection' ? 'Bridge Inspection' : view === 'nbi' ? 'NBI Ratings' : 'Structural Summary'}
          </h1>
          <div className="flex items-center gap-2">
            <button 
              onClick={simulateReportParsing}
              disabled={parsingActive}
              className="bg-slate-800 hover:bg-slate-700 text-slate-300 p-1.5 rounded-lg border border-slate-700 transition-all flex items-center gap-1 text-[9px] font-black uppercase"
            >
              {parsingActive ? <RotateCcw size={12} className="animate-spin" /> : <Upload size={12}/>}
              Import Legacy
            </button>
            <div className="flex bg-slate-800 p-1 rounded-lg text-[10px] font-bold border border-slate-700">
              <button onClick={() => setNomenclature(NOMENCLATURES.TXDOT)} className={`px-2 py-1 rounded ${nomenclature === NOMENCLATURES.TXDOT ? 'bg-blue-600 text-white shadow-sm' : 'text-slate-400'}`}>TX</button>
              <button onClick={() => setNomenclature(NOMENCLATURES.NCDOT)} className={`px-2 py-1 rounded ${nomenclature === NOMENCLATURES.NCDOT ? 'bg-blue-600 text-white shadow-sm' : 'text-slate-400'}`}>NC</button>
            </div>
          </div>
        </div>
        
        {(view === 'inspection' || view === 'nbi') && (
          <div className="flex flex-col gap-3">
             <button 
              onClick={toggleInspectionType}
              className={`w-full py-2.5 rounded-xl font-black text-xs uppercase tracking-widest flex items-center justify-center gap-3 border-2 transition-all shadow-lg active:scale-[0.98] ${inspectionType === INSPECTION_TYPES.TOPSIDE ? 'bg-blue-600 border-blue-500 text-white' : 'bg-slate-800 border-slate-700 text-blue-400'}`}
             >
               <RefreshCw size={16} />
               Active Module: {inspectionType === INSPECTION_TYPES.TOPSIDE ? 'Topside' : 'Underside'}
             </button>
          </div>
        )}
      </header>

      <main className="p-4 space-y-4 mx-auto w-full max-w-full">
        {view === 'inspection' && (
          <div className="max-w-2xl mx-auto w-full space-y-4">
            <section className="bg-white p-4 rounded-xl shadow-md border-t-4 border-slate-900">
              <h2 className="text-[11px] font-black text-slate-400 uppercase mb-3 tracking-widest text-center">Structural Stationing</h2>
              <label className="text-xs font-black text-slate-600 mb-1 block uppercase">Support / Span Designation</label>
              <select value={currentLocation} onChange={(e) => setCurrentLocation(e.target.value)} className="w-full bg-slate-50 border border-slate-200 rounded-lg p-3 text-lg font-black text-slate-900 outline-none focus:ring-2 ring-blue-500 transition-all">
                {locationSequence.map(loc => <option key={loc} value={loc}>{loc}</option>)}
              </select>
            </section>

            <section className="bg-white p-4 rounded-xl shadow-md border-t-4 border-blue-600 relative">
              {editId && <div className="absolute -top-3 right-4 flex items-center gap-1 text-[10px] font-black bg-blue-600 text-white px-3 py-1 rounded-full shadow-lg uppercase animate-pulse"><Edit2 size={12}/> Modifying Record</div>}

                <h2 className="text-lg font-black text-red-600 uppercase tracking-tighter flex items-center gap-2"><ShieldAlert size={24} className="animate-pulse"/> Critical Finding Summary</h2>
                <span className="bg-red-600 text-white px-4 py-1 rounded-full text-[10px] font-black uppercase tracking-widest">{criticalFindingsSummary.length} Active</span>
              </div>
              <div className="overflow-x-auto">
                 <table className="w-full text-left text-[12px] border-collapse">
                    <thead>
                      <tr className="border-b border-red-200 text-[10px] font-black text-red-600 uppercase bg-red-50/50">
                        <th className="py-3 px-4 w-48">Location/Element</th>
                        <th className="py-3 px-4">Structural Write-up</th>
                        <th className="py-3 px-4 text-right w-40">Quantity</th>
                      </tr>
                    </thead>
                    <tbody>
                      {criticalFindingsSummary.map(d => (
                        <tr key={d.id} className="border-b border-red-100 hover:bg-red-50/30">
                          <td className="py-4 px-4 align-top">
                            <div className="text-[11px] font-black text-slate-900 uppercase leading-none">{String(d.location)}</div>
                            <div className="text-[9px] font-bold text-red-600 uppercase mt-1">{String(d.element)}</div>
                          </td>
                          <td className="py-4 px-4 align-top">
                            <div className="text-[11px] font-black text-slate-800 uppercase">{String(d.defect)}</div>
                            <div className="text-[10px] font-medium text-slate-500 italic mt-1 leading-relaxed">{String(d.locationDesc)}</div>
                          </td>
                          <td className="py-4 px-4 text-right align-top font-black text-red-700 uppercase">
                             {String(d.quantity)}
                          </td>
                        </tr>
                      ))}
                      {criticalFindingsSummary.length === 0 && (
                        <tr><td colSpan={3} className="py-12 text-center text-[10px] font-black text-slate-300 uppercase italic">No Critical Deficiencies Reported</td></tr>
                      )}
                    </tbody>
                 </table>
              </div>
            </section>
          </div>
        )}
      </main>

      {/* MODAL: FORM 2598 (CIF) */}
      {showCIFModal && (
        <div className="fixed inset-0 bg-slate-900/90 backdrop-blur-sm z-[100] flex items-center justify-center p-4 overflow-y-auto">
          <div className="bg-white w-full max-w-4xl rounded-3xl shadow-2xl border-4 border-red-600 flex flex-col max-h-[95vh]">
            <div className="p-6 bg-red-600 text-white flex items-center justify-between shrink-0"><div className="flex items-center gap-3"><ShieldAlert size={32} className="animate-pulse" /><div><h2 className="text-xl font-black uppercase tracking-tighter leading-none">Critical Inspection Finding</h2><p className="text-[10px] font-bold opacity-80 uppercase tracking-widest mt-1">Form 2598 Implementation</p></div></div><button onClick={() => setShowCIFModal(false)} className="p-2 hover:bg-red-700 rounded-full transition-all"><X size={24}/></button></div>
            <div className="p-6 overflow-y-auto space-y-6">
               <div className="border-2 border-slate-200 p-8 rounded-xl bg-white shadow-inner font-serif">
                <div className="grid grid-cols-2 gap-x-8 gap-y-4 text-[11px] border-b-2 border-slate-900 pb-6 mb-6"><div><span className="font-bold block text-[8px] text-slate-500 uppercase">Structure No.</span><span className="font-black">{cifData.structureNumber}</span></div><div><span className="font-bold block text-[8px] text-slate-500 uppercase">Inspection Date</span><span className="font-black">{cifData.inspectionDate}</span></div></div>
                <div className="space-y-4"><div className="bg-slate-100 p-2 font-bold uppercase text-[9px] border border-slate-300">Findings / Reason for CIF:</div><textarea className="w-full font-black outline-none bg-yellow-50 p-2 text-[10px] border border-slate-300 min-h-[100px]" value={cifData.findings} onChange={(e)=>setCifData({...cifData, findings: e.target.value})} /><div className="bg-slate-100 p-2 font-bold uppercase text-[9px] border border-slate-300">Inspector's Recommendation to Owner:</div><textarea className="w-full font-black outline-none bg-yellow-50 p-2 text-[10px] border border-slate-300 min-h-[100px]" value={cifData.recommendation} onChange={(e)=>setCifData({...cifData, recommendation: e.target.value})} placeholder="Specify remedial strategies..." /></div>
                <div className="mt-8"><div className="flex items-center justify-between mb-2"><span className="text-[10px] font-black text-red-600 uppercase underline">Safety Evidence Media</span><label className="bg-red-600 text-white px-3 py-1 rounded text-[8px] font-black uppercase cursor-pointer flex items-center gap-1 hover:bg-red-700 shadow-sm"><Camera size={12}/> Append Evidence<input type="file" multiple accept="image/*" className="hidden" onChange={(e)=>handlePhotoUpload(e, 'cif')} /></label></div><div className="grid grid-cols-1 gap-4">
                    {cifData.photos.map((p, idx) => (<div key={idx} className="flex gap-4 bg-slate-50 p-3 rounded-xl border border-slate-200"><img src={p.url} className="w-24 h-24 rounded-lg object-cover border-2 border-white shadow-md" alt="evidence" /><div className="flex-1 space-y-2"><div className="flex gap-1 items-center mb-1"><input placeholder="Description..." value={p.description} onChange={(e) => updatePhotoDescription(idx, e.target.value, 'cif')} className="flex-1 bg-white border border-slate-300 rounded p-1 text-[10px] font-bold outline-none" /><button onClick={() => copyNoteToPhoto(idx, 'cif')} className="p-1.5 bg-red-50 text-red-600 rounded hover:bg-red-100 transition-colors"><Copy size={12}/></button></div></div><button onClick={() => removePhoto(idx, 'cif')} className="text-slate-300 hover:text-red-600 self-center"><X size={18}/></button></div>))}
                  </div></div>
                <div className="mt-8 p-4 border-2 border-red-500 rounded-lg bg-red-50/10"><div className="text-[10px] font-black text-red-600 uppercase mb-4 underline decoration-red-300 decoration-2">Safety Protocol Verification</div><div className="space-y-3"><label className="flex items-center gap-3 cursor-pointer group"><input type="checkbox" checked={cifData.phoneNotified} onChange={(e) => setCifData({...cifData, phoneNotified: e.target.checked})} className="w-5 h-5 accent-red-600" /><span className="text-[10px] font-black text-slate-700 uppercase group-hover:text-red-600">Immediate Phone Notification to District Bridge Office Complete</span></label><label className="flex items-center gap-3 cursor-pointer group"><input type="checkbox" checked={cifData.assetWiseLogged} onChange={(e) => setCifData({...cifData, assetWiseLogged: e.target.checked})} className="w-5 h-5 accent-blue-600" /><span className="text-[10px] font-black text-slate-700 uppercase group-hover:text-blue-600">AssetWise FUA Transitioned to "District Bridge Review"</span></label></div></div>
              </div>
            </div>
            <div className="p-6 bg-slate-50 border-t border-slate-200 shrink-0"><button onClick={completeCIF} className={`w-full py-4 rounded-2xl font-black text-sm uppercase tracking-widest transition-all ${(!cifData.phoneNotified || !cifData.assetWiseLogged) ? 'bg-slate-300 text-slate-500 cursor-not-allowed shadow-inner' : 'bg-red-600 text-white hover:bg-red-700 shadow-xl'}`}>Finalize CIF Safety Record</button></div>
          </div>
        </div>
      )}

      {/* MODAL: FOLLOW-UP ACTION (FUA) - MAINTENANCE */}
      {showFUAModal && (
        <div className="fixed inset-0 bg-slate-900/90 backdrop-blur-sm z-[100] flex items-center justify-center p-4 overflow-y-auto">
          <div className="bg-white w-full max-w-4xl rounded-3xl shadow-2xl border-4 border-blue-600 flex flex-col max-h-[95vh]">
            <div className="p-6 bg-blue-600 text-white flex items-center justify-between shrink-0"><div className="flex items-center gap-3"><Wrench size={32} /><div><h2 className="text-xl font-black uppercase tracking-tighter leading-none">Maintenance Follow-Up Action</h2><p className="text-[10px] font-bold opacity-80 uppercase tracking-widest mt-1">AssetWise Maintenance Module Sync</p></div></div><button onClick={() => setShowFUAModal(false)} className="p-2 hover:bg-blue-700 rounded-full transition-all"><X size={24}/></button></div>
            <div className="p-6 overflow-y-auto space-y-6">
              <div className="border-2 border-slate-200 p-6 rounded-xl bg-white shadow-inner font-sans relative overflow-hidden"><div className="grid grid-cols-2 gap-x-8 gap-y-4 text-[11px] mb-8"><div className="border-b border-slate-200 pb-1"><span className="text-[9px] font-bold text-slate-400 block uppercase">Priority Level (Rec Type)</span><select value={fuaData.priority} onChange={(e)=>setFuaData({...fuaData, priority: e.target.value})} className="w-full font-black outline-none bg-blue-50/50"><option value="Level 1">Level 1 (30 Days)</option><option value="Level 2">Level 2 (6 Months)</option><option value="Level 3">Level 3 (24 Months)</option><option value="Level 4">Level 4 (No Timeframe)</option></select></div><div className="border-b border-slate-200 pb-1"><span className="text-[9px] font-bold text-slate-400 block uppercase">Previously Recommended?</span><select value={fuaData.previouslyRecommended} onChange={(e)=>setFuaData({...fuaData, previouslyRecommended: e.target.value})} className="w-full font-black outline-none bg-blue-50/50"><option value="N">No</option><option value="Y">Yes</option></select></div></div><div className="space-y-4"><div><div className="text-[10px] font-black text-slate-800 uppercase mb-1">Description of Issue</div><textarea value={fuaData.description} onChange={(e)=>setFuaData({...fuaData, description: e.target.value})} className="w-full h-24 bg-blue-50/30 border border-slate-300 rounded p-3 text-xs font-bold" /></div><div><div className="text-[10px] font-black text-slate-800 uppercase mb-1">Remedial Recommendation</div><textarea value={fuaData.recommendation} onChange={(e)=>setFuaData({...fuaData, recommendation: e.target.value})} className="w-full h-24 bg-blue-50/30 border border-slate-300 rounded p-3 text-xs font-bold" /></div></div><div className="mt-8"><div className="flex items-center justify-between mb-2"><span className="text-[10px] font-black text-blue-600 uppercase underline">Remedial Evidence Manifest</span><label className="bg-blue-600 text-white px-3 py-1 rounded text-[8px] font-black uppercase cursor-pointer flex items-center gap-1 hover:bg-blue-700 shadow-sm"><Camera size={12}/> Append Evidence<input type="file" multiple accept="image/*" className="hidden" onChange={(e)=>handlePhotoUpload(e, 'fua')} /></label></div><div className="grid grid-cols-1 gap-4">
                    {fuaData.photos.map((p, idx) => (<div key={idx} className="flex gap-4 bg-slate-50 p-3 rounded-xl border border-slate-200"><img src={p.url} className="w-24 h-24 rounded-lg object-cover border-2 border-white shadow-md" alt="evidence" /><div className="flex-1 space-y-2"><div className="flex gap-1 items-center mb-1"><input placeholder="Description..." value={p.description} onChange={(e) => updatePhotoDescription(idx, e.target.value, 'fua')} className="flex-1 bg-white border border-slate-300 rounded p-1 text-[10px] font-bold outline-none" /><button onClick={() => copyNoteToPhoto(idx, 'fua')} className="p-1.5 bg-blue-50 text-blue-600 rounded hover:bg-blue-100 transition-colors"><Copy size={12}/></button></div></div><button onClick={() => removePhoto(idx, 'fua')} className="text-slate-300 hover:text-red-600 self-center"><X size={18}/></button></div>))}
                  </div></div>
                {(fuaData.priority === 'Level 1' || fuaData.priority === 'Level 2') && (<div className="mt-8 p-4 border-2 border-orange-500 rounded-lg bg-orange-50"><div className="flex items-center gap-2 text-orange-700 font-black text-[10px] uppercase mb-3"><Phone size={14}/> Urgent Notification Required</div><label className="flex items-center gap-3 cursor-pointer"><input type="checkbox" checked={fuaData.phoneNotified} onChange={(e)=>setFuaData({...fuaData, phoneNotified: e.target.checked})} className="w-5 h-6 accent-orange-600" /><span className="text-[10px] font-black text-slate-700 uppercase">Confirmed: District Phone Notification Logged</span></label></div>)}
                <div className="mt-4 p-4 border-2 border-blue-500 rounded-lg bg-blue-50"><label className="flex items-center gap-3 cursor-pointer"><input type="checkbox" checked={fuaData.assetWiseLogged} onChange={(e)=>setFuaData({...fuaData, assetWiseLogged: e.target.checked})} className="w-5 h-5 accent-blue-600" /><span className="text-[10px] font-black text-slate-700 uppercase">Synchronized with AssetWise Maintenance Module</span></label></div></div></div>
            <div className="p-6 bg-slate-50 border-t border-slate-200 shrink-0"><button onClick={completeFUA} className={`w-full py-4 rounded-2xl font-black text-sm uppercase tracking-widest transition-all ${((fuaData.priority === 'Level 1' || fuaData.priority === 'Level 2') && !fuaData.phoneNotified) || !fuaData.assetWiseLogged ? 'bg-slate-300 text-slate-500 cursor-not-allowed shadow-inner' : 'bg-blue-600 text-white hover:bg-blue-700 shadow-xl'}`}>Finalize Follow-Up Action Record</button></div>
          </div>
        </div>
      )}

      <nav className="fixed bottom-0 left-0 right-0 h-20 bg-white border-t border-slate-200 flex items-center justify-around z-50 shadow-[0_-10px_30px_-10px_rgba(0,0,0,0.15)] px-4">
        <button onClick={() => setView('inspection')} className={`flex-1 flex flex-col items-center justify-center transition-all h-full ${view === 'inspection' ? 'text-blue-600 border-t-4 border-blue-600 bg-blue-50/30 shadow-inner' : 'text-slate-300 hover:text-slate-500'}`}><Zap size={22}/><span className="text-[9px] font-black uppercase mt-1.5 tracking-widest">Element Data</span></button>
        <button onClick={() => setView('nbi')} className={`flex-1 flex flex-col items-center justify-center transition-all h-full ${view === 'nbi' ? 'text-blue-600 border-t-4 border-blue-600 bg-blue-50/30 shadow-inner' : 'text-slate-300 hover:text-slate-500'}`}><ClipboardList size={22}/><span className="text-[9px] font-black uppercase mt-1.5 tracking-widest">NBI Ratings</span></button>
        <button onClick={() => setView('summary')} className={`flex-1 flex flex-col items-center justify-center transition-all h-full ${view === 'summary' ? 'text-blue-600 border-t-4 border-blue-600 bg-blue-50/30 shadow-inner' : 'text-slate-300 hover:text-slate-500'}`}><BarChart3 size={22}/><span className="text-[9px] font-black uppercase mt-1.5 tracking-widest">Summary</span></button>
      </nav>
    </div>
  );
}
